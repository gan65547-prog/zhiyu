export const chatStream = async (
  data: ChatRequest,
  onMessage: (content: string) => void,
  onSources?: (sources: any[]) => void,
  onDone?: (info: any) => void
) => {
  const token = localStorage.getItem('token')
  const baseURL = import.meta.env.VITE_API_BASE_URL || '/api/v1'
  
  // 发送 POST 请求
  const response = await fetch(`${baseURL}/chat/stream`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${token}`
    },
    body: JSON.stringify(data)
  })
  
  // 读取 SSE 流式响应
  const reader = response.body?.getReader()
  const decoder = new TextDecoder()
  
  let buffer = ''
  
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    
    buffer += decoder.decode(value, { stream: true })
    const lines = buffer.split('\n')
    
    buffer = lines.pop() || ''
    
    for (const line of lines) {
      if (line.startsWith('data: ')) {
        try {
          const dataStr = line.slice(6).trim()
          if (!dataStr) continue
          
          const data = JSON.parse(dataStr)
          
          // 处理不同类型的消息
          if (data.delta !== undefined) {
            onMessage(data.delta)
          } else if (data.event === 'message' && data.data) {
            onMessage(data.data)
          } else if (data.event === 'sources' && data.data) {
            onSources?.(data.data)
          } else if (data.event === 'done') {
            onDone?.(data.data)
          }
        } catch (e) {
          console.error('Failed to parse SSE data:', e)
        }
      }
    }
  }
}
```

@router.post("/stream")
async def chat_stream(
    request: ChatRequest,
    chat_service: Annotated[ChatService, Depends(get_chat_service)],
    current_user: Annotated[dict, Depends(get_current_active_user)]
):
    """流式对话请求"""
    try:
        import json
        
        dataset_id = UUID(request.dataset_id)
        conversation_id = UUID(request.conversation_id) if request.conversation_id else None
        user_id = current_user["sub"]
        tenant_id = current_user.get("tenant_id")
        
        # 验证dataset和workspace权限
        await chat_service.verify_dataset_access(
            dataset_id=dataset_id,
            user_id=user_id,
            tenant_id=tenant_id
        )

        async def generate():
            async for chunk in chat_service.chat_stream(
                dataset_id=dataset_id,
                query=request.query,
                conversation_id=conversation_id,
                history=[msg.model_dump() for msg in request.history] if request.history else None,
                retrieval_mode=request.retrieval_mode.value,
                top_k=request.top_k,
                temperature=request.temperature,
                max_tokens=request.max_tokens
            ):
                yield f"data: {json.dumps(chunk)}\n\n"

        return StreamingResponse(
            generate(),
            media_type="text/event-stream",
            headers={"Cache-Control": "no-cache"}
        )
    except Exception as e:
        # 错误处理...

async def chat_stream(
    self,
    dataset_id: UUID,
    query: str,
    conversation_id: Optional[UUID] = None,
    history: Optional[List[Dict[str, str]]] = None,
    retrieval_mode: str = "hybrid",
    top_k: int = 5,
    temperature: float = 0.7,
    max_tokens: int = 2000
) -> AsyncIterator[dict]:
    """流式对话请求"""
    if not conversation_id:
        conv = await self.create_conversation(
            dataset_id=dataset_id,
            title=query[:50] if len(query) > 50 else query
        )
        conversation_id = UUID(conv["id"])

    response_text = ""

    try:
        # 获取数据集的工作空间ID
        from infrastructure.database.models import Dataset
        from sqlalchemy import select
        result = await self.session.execute(
            select(Dataset.workspace_id).where(Dataset.id == str(dataset_id))
        )
        workspace_id = result.scalar_one_or_none()
        
        if not workspace_id:
            raise ValueError(f"数据集 {dataset_id} 不存在")
        
        # 使用内存向量存储
        from core.rag.vector_store import MemoryVectorStore
        from core.rag.embedding import get_embedding_service
        vector_store = MemoryVectorStore.get_instance(
            collection_name=f"dataset_{str(dataset_id)}",
            embedding_service=get_embedding_service()
        )
        
        # 创建RAG引擎
        from core.rag.rag_engine import RAGEngineImpl
        rag_engine = RAGEngineImpl(vector_store=vector_store)
        
        # 创建查询上下文
        from core.interfaces.rag_engine import QueryContext
        query_context = QueryContext(
            workspace_id=UUID(workspace_id),
            dataset_ids=[dataset_id],
            retrieval_mode=retrieval_mode,
            top_k=top_k
        )
        
        # 执行RAG查询（流式）
        async for chunk in rag_engine.query_stream(
            query=query,
            context=query_context,
            conversation_history=history,
            temperature=temperature,
            max_tokens=max_tokens
        ):
            if chunk.get("type") == "token":
                delta = chunk.get("delta", "")
                response_text += delta
                yield {
                    "delta": delta,
                    "conversation_id": str(conversation_id)
                }
    except Exception as e:
        # 错误处理...
        
    # 保存消息到数据库
    await self._save_message(...)
```

### 5. RAG 引擎层 - `core/rag/rag_engine.py`

**关键函数：** `query_stream()` (第101行)

```python
async def query_stream(
    self,
    query: str,
    context: QueryContext,
    conversation_history: Optional[List[Dict[str, str]]] = None,
    temperature: float = 0.7,
    max_tokens: int = 2000
):
    """流式执行RAG查询"""
    retrieval_start = time.time()

    # 1. 检索阶段 - 检索相关文档
    documents = await self.retrieve(query, context)

    retrieval_time = int((time.time() - retrieval_start) * 1000)

    yield {
        "type": "retrieval",
        "retrieved_count": len(documents),
        "retrieval_time_ms": retrieval_time
    }

    # 2. 生成阶段 - 流式生成回答
    async for token in self.response_generator.generate_stream(
        query=query,
        documents=documents,
        conversation_history=conversation_history,
        temperature=temperature,
        max_tokens=max_tokens
    ):
        yield {"type": "token", "delta": token}
```

**检索逻辑 - `retrieve()` (第131行)

```python
async def retrieve(
    self,
    query: str,
    context: QueryContext
) -> List[RetrievedDocument]:
    """检索相关文档"""
    if context.retrieval_mode == "semantic":
        return await self.semantic_search(query, context)
    elif context.retrieval_mode == "keyword":
        return await self.keyword_search(query, context)
    else:
        semantic_results = await self.semantic_search(query, context)
        keyword_results = await self.keyword_search(query, context)
        merged = self._merge_results(semantic_results, keyword_results)
        return merged
```

```python
def _build_messages(
    self,
    query: str,
    documents: List[RetrievedDocument],
    conversation_history: Optional[List[Dict[str, str]]] = None
) -> List[Dict[str, str]]:
    """构建消息列表"""
    messages = []
    
    # 系统提示
    system_prompt = """你是一个专业的AI助手。请根据提供的上下文信息回答用户的问题。
回答时请遵循以下原则：
1. 基于上下文信息给出准确、详细的回答
2. 如果上下文中没有相关信息，请明确说明
3. 回答时注明信息来源
4. 保持回答的专业性和准确性"""
    
    messages.append({"role": "system", "content": system_prompt})
    
    # 添加对话历史
    if conversation_history:
        for msg in conversation_history[-5:]:
            role = msg.get("role", "user")
            content = msg.get("content", "")
            if role == "user":
                messages.append({"role": "user", "content": content})
            elif role == "assistant":
                messages.append({"role": "assistant", "content": content})
    
    # 构建上下文
    context_parts = []
    for i, doc in enumerate(documents, 1):
        context_parts.append(
            f"[文档{i}]\n{doc.content}\n来源: {doc.document_name}"
        )
    
    context = "\n\n".join(context_parts) if context_parts else "无相关上下文"
    
    # 添加当前问题
    user_message = f"""[上下文信息]
{context}

[用户问题]
{query}"""
    
    messages.append({"role": "user", "content": user_message})
    
    return messages
```

## 数据流转图

```
用户输入 
  ↓
ChatView.vue (handleSend)
  ↓
chat.ts (chatStream) → POST /api/v1/chat/stream
  ↓
chat.py (chat_stream 路由)
  ↓
chat_service.py (chat_stream)
  ↓
rag_engine.py (query_stream)
  ├─→ retrieve() → 向量检索相关文档
  └─→ generate_stream()
       ↓
llm_provider.py (generate_stream)
  ├─→ _build_messages() → 构建包含上下文的提示
  └─→ 调用 DeepSeek/Ollama API
       ↓
流式返回 token → 前端逐字显示
  ↓
保存消息到数据库
```
